# Fine-Tuning Revamp Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build three ≥10K fine-tuning corpora (extraction, classification, embedding) plus a document classifier stub, then train and evaluate 9 specialist models against a qwen3.5:4b prompted baseline and deploy per-purpose winners.

**Architecture:** A canonical JSON schema anchors a 7K shared corpus annotated three ways, plus 3K per-purpose top-ups targeting failure modes from prior evals. Generation runs across many Claude Code sessions via `HANDOFF.md`. Mechanical validation gates every batch. Converters transform the canonical schema into each model family's native training format. Pretrained qwen3.5:4b participates in every eval as a prompted baseline, with QLoRA training deferred until the baseline shows a fine-tuneable gap.

**Tech Stack:** Python 3.11, PyTorch + Hugging Face Transformers + PEFT, GLiNER, REBEL (BART), Sentence Transformers, Ollama (for qwen3.5:4b), YAML config, pytest.

**Spec:** `docs/superpowers/specs/2026-04-14-fine-tuning-revamp-design.md`

```yaml
    You are a memory classifier. Classify the input memory into exactly one of seven memory types.
```

```python
# fine-tuning/scripts/validate_v2.py
"""Mechanical validator for v2 fine-tuning corpora.

Validates shared rows, extraction top-up rows, classification top-up rows,
and retrieval top-up rows against the schema defined in the design spec.
No model calls — pure schema + offset + taxonomy membership checks.
"""
from __future__ import annotations

import argparse
import json
import sys
from dataclasses import dataclass, field
from difflib import SequenceMatcher
from pathlib import Path
from typing import Iterable

import yaml

ROOT = Path(__file__).resolve().parents[1]
CONFIG_PATH = ROOT / "config" / "taxonomy_v2.yaml"


@dataclass
class ValidationResult:
    ok: bool
    errors: list[str] = field(default_factory=list)


def load_taxonomy(path: Path = CONFIG_PATH) -> dict:
    with path.open("r", encoding="utf-8") as f:
        data = yaml.safe_load(f)
    # Flatten entity_types across domains into a single set for fast membership
    entity_types: set[str] = set()
    for types in data["entity_types"].values():
        entity_types.update(types)
    relation_types = set(data["relation_types"]["core"]) | set(data["relation_types"]["extended"])
    data["_entity_type_set"] = entity_types
    data["_relation_type_set"] = relation_types
    data["_memory_type_set"] = set(data["memory_types"])
    data["_allowed_domains"] = {"technical", "ai_coding", "personal", "knowledge", "cross", "noise"}
    return data


def _ratio(a: str, b: str) -> float:
    return SequenceMatcher(None, a, b).ratio()


def _validate_entities(row: dict, taxonomy: dict) -> list[str]:
    errs: list[str] = []
    text = row["text"]
    for i, ent in enumerate(row.get("entities", [])):
        required = {"name", "type", "start", "end"}
        missing = required - ent.keys()
        if missing:
            errs.append(f"entity[{i}] missing fields: {sorted(missing)}")
            continue
        if ent["type"] not in taxonomy["_entity_type_set"]:
            errs.append(f"entity[{i}] unknown entity type: {ent['type']}")
        start, end = ent["start"], ent["end"]
        if not isinstance(start, int) or not isinstance(end, int):
            errs.append(f"entity[{i}] start/end must be int")
            continue
        if start < 0 or end > len(text) or start >= end:
            errs.append(f"entity[{i}] bad offset range [{start},{end}] for text length {len(text)}")
            continue
        if text[start:end] != ent["name"]:
            errs.append(
                f"entity[{i}] offset mismatch: text[{start}:{end}]={text[start:end]!r} != name={ent['name']!r}"
            )
    return errs


def _validate_relationships(row: dict, taxonomy: dict) -> list[str]:
    errs: list[str] = []
    entity_names = {e["name"] for e in row.get("entities", [])}
    for i, rel in enumerate(row.get("relationships", [])):
        required = {"source", "target", "type"}
        missing = required - rel.keys()
        if missing:
            errs.append(f"relationship[{i}] missing fields: {sorted(missing)}")
            continue
        if rel["source"] not in entity_names:
            errs.append(f"relationship[{i}] source endpoint not in entity list: {rel['source']}")
        if rel["target"] not in entity_names:
            errs.append(f"relationship[{i}] target endpoint not in entity list: {rel['target']}")
        if rel["type"] not in taxonomy["_relation_type_set"]:
            errs.append(f"relationship[{i}] unknown relation type: {rel['type']}")
    return errs


def _validate_queries(row: dict, taxonomy: dict) -> list[str]:
    errs: list[str] = []
    queries = row.get("queries")
    if queries is None:
        return errs   # queries are optional on shared rows only when omitted explicitly (noise)
    if len(queries) != 2:
        errs.append(f"queries must have exactly 2 items, got {len(queries)}")
        return errs
    forms = {q.get("form") for q in queries}
    if forms != {"question", "keyword"}:
        errs.append(f"queries must have forms {{question, keyword}}, got {forms}")
    text = row["text"]
    max_q_sim = taxonomy["validator"]["max_query_similarity_to_source"]
    max_pair_sim = taxonomy["validator"]["max_query_pair_similarity"]
    for i, q in enumerate(queries):
        if _ratio(q["text"], text) >= max_q_sim:
            errs.append(f"queries[{i}] too similar to source (verbatim quote) — normalized ratio >= {max_q_sim}")
    if len(queries) == 2 and _ratio(queries[0]["text"], queries[1]["text"]) >= max_pair_sim:
        errs.append(f"queries[0] and queries[1] too similar — normalized ratio >= {max_pair_sim}")
    return errs


def validate_shared_row(row: dict, taxonomy: dict) -> ValidationResult:
    errs: list[str] = []
    required = {"id", "domain", "text", "memory_type", "noise", "entities", "relationships"}
    missing = required - row.keys()
    if missing:
        return ValidationResult(ok=False, errors=[f"missing required fields: {sorted(missing)}"])
    if row["domain"] not in taxonomy["_allowed_domains"]:
        errs.append(f"unknown domain: {row['domain']}")
    if row["memory_type"] not in taxonomy["_memory_type_set"]:
        errs.append(f"unknown memory_type: {row['memory_type']}")
    errs += _validate_entities(row, taxonomy)
    errs += _validate_relationships(row, taxonomy)
    if not row["noise"] and not row["entities"]:
        errs.append("non-noise shared row has zero entities")
    if "queries" in row:
        errs += _validate_queries(row, taxonomy)
    return ValidationResult(ok=not errs, errors=errs)


_EXTRACTION_ALLOWED = {"id", "domain", "text", "memory_type", "entities", "relationships"}
_CLASSIFICATION_ALLOWED = {"id", "domain", "text", "memory_type", "pair_id"}
_RETRIEVAL_ALLOWED = {"id", "source_id", "query", "positive", "negative", "query_kind"}


def _foreign_fields(row: dict, allowed: set[str]) -> list[str]:
    foreign = set(row.keys()) - allowed
    return [f"foreign field on top-up row: {k}" for k in sorted(foreign)]


def validate_extraction_topup_row(row: dict, taxonomy: dict) -> ValidationResult:
    errs = _foreign_fields(row, _EXTRACTION_ALLOWED)
    required = {"id", "domain", "text", "memory_type", "entities", "relationships"}
    missing = required - row.keys()
    if missing:
        errs.append(f"missing required fields: {sorted(missing)}")
        return ValidationResult(ok=False, errors=errs)
    if row["memory_type"] not in taxonomy["_memory_type_set"]:
        errs.append(f"unknown memory_type: {row['memory_type']}")
    errs += _validate_entities(row, taxonomy)
    errs += _validate_relationships(row, taxonomy)
    return ValidationResult(ok=not errs, errors=errs)


def validate_classification_topup_row(row: dict, taxonomy: dict) -> ValidationResult:
    errs = _foreign_fields(row, _CLASSIFICATION_ALLOWED)
    required = {"id", "domain", "text", "memory_type"}
    missing = required - row.keys()
    if missing:
        errs.append(f"missing required fields: {sorted(missing)}")
        return ValidationResult(ok=False, errors=errs)
    if row["memory_type"] not in taxonomy["_memory_type_set"]:
        errs.append(f"unknown memory_type: {row['memory_type']}")
    return ValidationResult(ok=not errs, errors=errs)


def validate_retrieval_topup_row(row: dict, taxonomy: dict) -> ValidationResult:
    errs = _foreign_fields(row, _RETRIEVAL_ALLOWED)
    required = {"id", "source_id", "query", "query_kind"}
    missing = required - row.keys()
    if missing:
        errs.append(f"missing required fields: {sorted(missing)}")
        return ValidationResult(ok=False, errors=errs)
    allowed_kinds = {"paraphrase", "acronym", "typo", "code-switch", "multi-sentence", "hard-negative"}
    if row["query_kind"] not in allowed_kinds:
        errs.append(f"unknown query_kind: {row['query_kind']}")
    if row["query_kind"] == "hard-negative":
        if "negative" not in row:
            errs.append("hard-negative row requires 'negative' field")
    else:
        if "positive" not in row:
            errs.append("non-hard-negative row requires 'positive' field")
    return ValidationResult(ok=not errs, errors=errs)


def validate_file(path: Path, kind: str, taxonomy: dict) -> tuple[int, int, list[tuple[int, list[str]]]]:
    """Return (total, passed, failures). kind: shared|extraction|classification|retrieval."""
    validators = {
        "shared": validate_shared_row,
        "extraction": validate_extraction_topup_row,
        "classification": validate_classification_topup_row,
        "retrieval": validate_retrieval_topup_row,
    }
    validator = validators[kind]
    total = 0
    passed = 0
    failures: list[tuple[int, list[str]]] = []
    with path.open("r", encoding="utf-8") as f:
        for i, line in enumerate(f, start=1):
            line = line.strip()
            if not line:
                continue
            total += 1
            try:
                row = json.loads(line)
            except json.JSONDecodeError as e:
                failures.append((i, [f"invalid JSON: {e}"]))
                continue
            result = validator(row, taxonomy)
            if result.ok:
                passed += 1
            else:
                failures.append((i, result.errors))
    return total, passed, failures


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("path", type=Path)
    parser.add_argument("--kind", required=True, choices=["shared", "extraction", "classification", "retrieval"])
    parser.add_argument("--failed-dir", type=Path, default=None)
    args = parser.parse_args()

    taxonomy = load_taxonomy()
    total, passed, failures = validate_file(args.path, args.kind, taxonomy)
    print(f"{args.path}: {passed}/{total} passed, {len(failures)} failed")
    if failures and args.failed_dir:
        args.failed_dir.mkdir(parents=True, exist_ok=True)
        out = args.failed_dir / f"{args.path.stem}.failures.jsonl"
        with out.open("w", encoding="utf-8") as f:
            for line_no, errs in failures:
                f.write(json.dumps({"line": line_no, "errors": errs}) + "\n")
        print(f"wrote failures to {out}")
    return 0 if not failures else 1


if __name__ == "__main__":
    sys.exit(main())
```

```python
# fine-tuning/scripts/audit_existing.py
"""Sample and score existing corpora against the v2 schema.

Usage:
    python -m scripts.audit_existing \
        --source data/validated/all.jsonl --kind shared --out reports/audit_extraction.json
"""
from __future__ import annotations

import argparse
import json
import random
import sys
from dataclasses import dataclass, field
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))

from scripts.validate_v2 import (
    load_taxonomy,
    validate_shared_row,
    validate_extraction_topup_row,
    validate_classification_topup_row,
    validate_retrieval_topup_row,
)


@dataclass
class SampleResult:
    total: int
    passed: int
    failures: list[tuple[int, list[str]]] = field(default_factory=list)

    @property
    def pass_rate(self) -> float:
        return self.passed / self.total if self.total else 0.0


def sample_jsonl(path: Path, size: int, seed: int) -> list[dict]:
    rng = random.Random(seed)
    rows: list[dict] = []
    with path.open("r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                rows.append(json.loads(line))
    if size >= len(rows):
        return rows
    return rng.sample(rows, size)


def score_sample(sample: list[dict], kind: str) -> SampleResult:
    taxonomy = load_taxonomy()
    validators = {
        "shared": validate_shared_row,
        "extraction": validate_extraction_topup_row,
        "classification": validate_classification_topup_row,
        "retrieval": validate_retrieval_topup_row,
    }
    validator = validators[kind]
    total = len(sample)
    passed = 0
    failures: list[tuple[int, list[str]]] = []
    for i, row in enumerate(sample):
        result = validator(row, taxonomy)
        if result.ok:
            passed += 1
        else:
            failures.append((i, result.errors))
    return SampleResult(total=total, passed=passed, failures=failures)


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--source", type=Path, required=True)
    parser.add_argument("--kind", required=True, choices=["shared", "extraction", "classification", "retrieval"])
    parser.add_argument("--size", type=int, default=200)
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()

    sample = sample_jsonl(args.source, size=args.size, seed=args.seed)
    result = score_sample(sample, kind=args.kind)
    args.out.parent.mkdir(parents=True, exist_ok=True)
    with args.out.open("w", encoding="utf-8") as f:
        json.dump(
            {
                "source": str(args.source),
                "kind": args.kind,
                "total": result.total,
                "passed": result.passed,
                "pass_rate": result.pass_rate,
                "failures_head": result.failures[:20],
            },
            f,
            indent=2,
        )
    print(f"{args.source}: {result.passed}/{result.total} passed ({result.pass_rate:.2%})")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# fine-tuning/scripts/build_splits.py
"""Build deterministic train/val/test splits for v2 corpora.

Stratifies by `domain` for extraction/classification. For retrieval,
expands each shared row into 2 (query, positive) rows and appends the
retrieval top-up, splitting 90/10 train/val (test is the fixed 270-query set).
"""
from __future__ import annotations

import argparse
import json
import random
import sys
from collections import defaultdict
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]


def _read_jsonl(path: Path) -> list[dict]:
    if not path.exists():
        return []
    rows: list[dict] = []
    with path.open("r", encoding="utf-8") as f:
        for line in f:
            line = line.strip()
            if line:
                rows.append(json.loads(line))
    return rows


def _write_jsonl(path: Path, rows: list[dict]) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        for r in rows:
            f.write(json.dumps(r, ensure_ascii=False) + "\n")


def _stratified_split(rows: list[dict], key: str, seed: int, val_fraction: float, test_fraction: float) -> tuple[list[dict], list[dict], list[dict]]:
    rng = random.Random(seed)
    by_key: dict[str, list[dict]] = defaultdict(list)
    for r in rows:
        by_key[r.get(key, "_unknown")].append(r)
    train: list[dict] = []
    val: list[dict] = []
    test: list[dict] = []
    for k, items in sorted(by_key.items()):
        items = items[:]
        rng.shuffle(items)
        n = len(items)
        n_val = max(1, int(n * val_fraction)) if n >= 10 else 0
        n_test = max(1, int(n * test_fraction)) if n >= 10 else 0
        test += items[:n_test]
        val += items[n_test : n_test + n_val]
        train += items[n_test + n_val :]
    rng.shuffle(train)
    rng.shuffle(val)
    rng.shuffle(test)
    return train, val, test


def build_extraction_splits(shared_path: Path, topup_path: Path, out_dir: Path, seed: int = 42) -> None:
    shared = _read_jsonl(shared_path)
    topup = _read_jsonl(topup_path)
    # Project shared rows into extraction schema (drop queries)
    shared_ext = [
        {k: v for k, v in r.items() if k in {"id", "domain", "text", "memory_type", "entities", "relationships"}}
        for r in shared
    ]
    all_rows = shared_ext + topup
    train, val, test = _stratified_split(all_rows, key="domain", seed=seed, val_fraction=0.1, test_fraction=0.1)
    _write_jsonl(out_dir / "extraction_train.jsonl", train)
    _write_jsonl(out_dir / "extraction_val.jsonl", val)
    _write_jsonl(out_dir / "extraction_test.jsonl", test)


def build_classification_splits(shared_path: Path, topup_path: Path, out_dir: Path, seed: int = 42) -> None:
    shared = _read_jsonl(shared_path)
    topup = _read_jsonl(topup_path)
    shared_cls = [
        {k: v for k, v in r.items() if k in {"id", "domain", "text", "memory_type"}}
        for r in shared
    ]
    all_rows = shared_cls + topup
    train, val, test = _stratified_split(all_rows, key="memory_type", seed=seed, val_fraction=0.1, test_fraction=0.1)
    _write_jsonl(out_dir / "classification_train.jsonl", train)
    _write_jsonl(out_dir / "classification_val.jsonl", val)
    _write_jsonl(out_dir / "classification_test.jsonl", test)


def build_retrieval_splits(shared_path: Path, topup_path: Path, out_dir: Path, seed: int = 42, val_fraction: float = 0.1) -> None:
    shared = _read_jsonl(shared_path)
    topup = _read_jsonl(topup_path)
    pairs: list[dict] = []
    for row in shared:
        if row.get("noise"):
            continue
        for q in row.get("queries", []):
            pairs.append({
                "id": f"{row['id']}-{q['form']}",
                "source_id": row["id"],
                "query": q["text"],
                "positive": row["text"],
                "query_kind": q["form"],
            })
    pairs += topup
    rng = random.Random(seed)
    rng.shuffle(pairs)
    n_val = int(len(pairs) * val_fraction)
    val = pairs[:n_val]
    train = pairs[n_val:]
    _write_jsonl(out_dir / "retrieval_train.jsonl", train)
    _write_jsonl(out_dir / "retrieval_val.jsonl", val)


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--shared", type=Path, default=ROOT / "data/v2/shared/shared.jsonl")
    parser.add_argument("--extraction-topup", type=Path, default=ROOT / "data/v2/extraction/topup.jsonl")
    parser.add_argument("--classification-topup", type=Path, default=ROOT / "data/v2/classification/topup.jsonl")
    parser.add_argument("--retrieval-topup", type=Path, default=ROOT / "data/v2/retrieval/topup.jsonl")
    parser.add_argument("--out", type=Path, default=ROOT / "data/v2/splits")
    parser.add_argument("--seed", type=int, default=42)
    args = parser.parse_args()

    build_extraction_splits(args.shared, args.extraction_topup, args.out, seed=args.seed)
    build_classification_splits(args.shared, args.classification_topup, args.out, seed=args.seed)
    build_retrieval_splits(args.shared, args.retrieval_topup, args.out, seed=args.seed)
    print(f"splits written to {args.out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# fine-tuning/scripts/convert_to_gliner.py
"""Canonical → GLiNER-native format using the base model's tokenizer.

GLiNER's `Trainer` accepts rows in the form
{"tokenized_text": [...], "ner": [[start_tok, end_tok, label], ...]}

Rows whose character-level spans don't map cleanly to token boundaries
are written to failed_dir with a reason.
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


def _char_to_token_index(offsets: list[tuple[int, int]], char_idx: int, *, side: str) -> int | None:
    """Return token index whose character span contains char_idx. Side: 'start' or 'end'."""
    for ti, (cs, ce) in enumerate(offsets):
        if cs == ce:
            continue
        if side == "start" and cs <= char_idx < ce:
            return ti
        if side == "end" and cs < char_idx <= ce:
            return ti
    return None


def convert_file(src: Path, out: Path, *, failed_dir: Path, base_model: str) -> dict:
    from transformers import AutoTokenizer
    tok = AutoTokenizer.from_pretrained(base_model)
    failed_dir.mkdir(parents=True, exist_ok=True)
    failed_path = failed_dir / (src.stem + ".failures.jsonl")
    out.parent.mkdir(parents=True, exist_ok=True)

    converted = 0
    failed = 0
    with (
        src.open("r", encoding="utf-8") as fi,
        out.open("w", encoding="utf-8") as fo,
        failed_path.open("w", encoding="utf-8") as ff,
    ):
        for line_no, line in enumerate(fi, start=1):
            line = line.strip()
            if not line:
                continue
            row = json.loads(line)
            text = row["text"]
            enc = tok(text, return_offsets_mapping=True, add_special_tokens=False)
            offsets = enc["offset_mapping"]
            tokens = tok.convert_ids_to_tokens(enc["input_ids"])
            ner: list[list] = []
            ok = True
            for ent in row.get("entities", []):
                start_tok = _char_to_token_index(offsets, ent["start"], side="start")
                end_tok = _char_to_token_index(offsets, ent["end"], side="end")
                if start_tok is None or end_tok is None or end_tok < start_tok:
                    ok = False
                    ff.write(json.dumps({"line": line_no, "entity": ent, "reason": "boundary misalignment"}) + "\n")
                    break
                ner.append([start_tok, end_tok, ent["type"]])
            if not ok:
                failed += 1
                continue
            fo.write(json.dumps({"tokenized_text": tokens, "ner": ner}) + "\n")
            converted += 1
    return {"converted": converted, "failed": failed}


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("src", type=Path)
    p.add_argument("out", type=Path)
    p.add_argument("--failed-dir", type=Path, required=True)
    p.add_argument("--base-model", default="urchade/gliner_multitask-large-v0.5")
    a = p.parse_args()
    stats = convert_file(a.src, a.out, failed_dir=a.failed_dir, base_model=a.base_model)
    print(f"converted={stats['converted']} failed={stats['failed']}")
    total = stats["converted"] + stats["failed"]
    if total and stats["failed"] / total > 0.01:
        print("ERROR: failure rate exceeds 1%; training should not proceed", file=sys.stderr)
        return 2
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Canonical → REBEL linearized triplet format."""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


def _format_input(text: str, entities: list[dict]) -> str:
    entity_str = ", ".join(f"{e['name']} ({e['type']})" for e in entities)
    return f"{text} | Entities: {entity_str}"


def _format_output(relationships: list[dict]) -> str:
    if not relationships:
        return "<triplet> NONE"
    parts = [f"<triplet> {r['source']} <subj> {r['target']} <obj> {r['type']}" for r in relationships]
    return " ".join(parts)


def convert_file(src: Path, out: Path) -> int:
    out.parent.mkdir(parents=True, exist_ok=True)
    count = 0
    with src.open("r", encoding="utf-8") as fi, out.open("w", encoding="utf-8") as fo:
        for line in fi:
            line = line.strip()
            if not line:
                continue
            row = json.loads(line)
            fo.write(json.dumps({
                "input": _format_input(row["text"], row.get("entities", [])),
                "output": _format_output(row.get("relationships", [])),
            }) + "\n")
            count += 1
    return count


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("src", type=Path)
    p.add_argument("out", type=Path)
    a = p.parse_args()
    print(f"wrote {convert_file(a.src, a.out)} rows to {a.out}")
    return 0
```

```python
# fine-tuning/scripts/eval_baseline_qwen.py
"""qwen3.5:4b prompted baseline runner for all three tasks."""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path

import yaml
import requests

ROOT = Path(__file__).resolve().parents[1]
PROMPTS_PATH = ROOT / "config" / "prompts_v2.yaml"
TAXONOMY_PATH = ROOT / "config" / "taxonomy_v2.yaml"


def load_prompts() -> dict:
    with PROMPTS_PATH.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def load_taxonomy() -> dict:
    with TAXONOMY_PATH.open("r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def format_entity_list(entities: list[dict]) -> str:
    return "\n".join(f"- {e['name']} ({e['type']})" for e in entities)


def call_ollama(system: str, user: str, model: str, options: dict, format: str = "json") -> str:
    r = requests.post(
        "http://localhost:11434/api/chat",
        json={
            "model": model,
            "messages": [{"role": "system", "content": system}, {"role": "user", "content": user}],
            "options": options,
            "format": format,
            "stream": False,
        },
        timeout=120,
    )
    r.raise_for_status()
    return r.json()["message"]["content"]


def run_entity(text: str, prompts: dict, entity_types: list[str], model: str = "qwen3.5:4b", options: dict | None = None) -> list[dict]:
    options = options or {"temperature": 0.0, "seed": 42}
    system = prompts["system"].replace("{ENTITY_TYPES}", ", ".join(entity_types))
    user = prompts["user_template"].format(text=text)
    raw = call_ollama(system, user, model=model, options=options)
    try:
        data = json.loads(raw)
        return data.get("entities", [])
    except json.JSONDecodeError:
        return []


def run_relation(text: str, entities: list[dict], prompts: dict, relation_types: list[str], model: str = "qwen3.5:4b", options: dict | None = None) -> list[dict]:
    options = options or {"temperature": 0.0, "seed": 42}
    system = prompts["system"].replace("{RELATION_TYPES}", ", ".join(relation_types))
    user = prompts["user_template"].format(text=text, entity_list=format_entity_list(entities))
    raw = call_ollama(system, user, model=model, options=options)
    try:
        data = json.loads(raw)
        return data.get("relationships", [])
    except json.JSONDecodeError:
        return []


def run_classification(text: str, prompts: dict, model: str = "qwen3.5:4b", options: dict | None = None) -> str:
    options = options or {"temperature": 0.0, "seed": 42}
    user = prompts["user_template"].format(text=text)
    raw = call_ollama(prompts["system"], user, model=model, options=options)
    try:
        data = json.loads(raw)
        return data.get("memory_type", "")
    except json.JSONDecodeError:
        return ""


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--task", required=True, choices=["entity", "relation", "classification"])
    p.add_argument("--input", type=Path, required=True, help="test split JSONL")
    p.add_argument("--output", type=Path, required=True, help="predictions JSONL")
    args = p.parse_args()

    prompts_all = load_prompts()
    taxonomy = load_taxonomy()
    model = prompts_all["ollama"]["model"]
    options = prompts_all["ollama"]["options"]
    entity_types = [t for group in taxonomy["entity_types"].values() for t in group]
    relation_types = taxonomy["relation_types"]["core"] + taxonomy["relation_types"]["extended"]
    task_prompts = prompts_all[args.task if args.task != "classification" else "classification"]

    args.output.parent.mkdir(parents=True, exist_ok=True)
    with args.input.open("r", encoding="utf-8") as fi, args.output.open("w", encoding="utf-8") as fo:
        for line in fi:
            line = line.strip()
            if not line:
                continue
            row = json.loads(line)
            if args.task == "entity":
                pred = run_entity(row["text"], task_prompts, entity_types, model=model, options=options)
                fo.write(json.dumps({"id": row["id"], "entities": pred}) + "\n")
            elif args.task == "relation":
                pred = run_relation(row["text"], row.get("entities", []), task_prompts, relation_types, model=model, options=options)
                fo.write(json.dumps({"id": row["id"], "relationships": pred}) + "\n")
            else:
                pred = run_classification(row["text"], task_prompts, model=model, options=options)
                fo.write(json.dumps({"id": row["id"], "memory_type": pred}) + "\n")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# fine-tuning/scripts/eval_entity.py
"""Entity eval harness. Reports per-type P/R/F1, type accuracy, span IoU, latency."""
from __future__ import annotations

import argparse
import json
import sys
import time
from collections import defaultdict
from pathlib import Path


def _iou(a: tuple[int, int], b: tuple[int, int]) -> float:
    s = max(a[0], b[0])
    e = min(a[1], b[1])
    inter = max(0, e - s)
    union = (a[1] - a[0]) + (b[1] - b[0]) - inter
    return inter / union if union else 0.0


def score(gold_rows: list[dict], pred_rows: dict) -> dict:
    tp_by_type: dict[str, int] = defaultdict(int)
    fp_by_type: dict[str, int] = defaultdict(int)
    fn_by_type: dict[str, int] = defaultdict(int)
    ious: list[float] = []
    type_correct = 0
    type_total = 0
    for g in gold_rows:
        p = pred_rows.get(g["id"], {}).get("entities", [])
        gold_set = {(e["name"], e["type"], e["start"], e["end"]) for e in g["entities"]}
        pred_set = {(e["name"], e["type"], e["start"], e["end"]) for e in p}
        for e in gold_set:
            if e in pred_set:
                tp_by_type[e[1]] += 1
            else:
                fn_by_type[e[1]] += 1
        for e in pred_set:
            if e not in gold_set:
                fp_by_type[e[1]] += 1
        # span IoU + type accuracy on best-match pairs
        for ge in g["entities"]:
            best = max(
                (pe for pe in p if pe["name"] == ge["name"]),
                key=lambda pe: _iou((ge["start"], ge["end"]), (pe["start"], pe["end"])),
                default=None,
            )
            if best is not None:
                ious.append(_iou((ge["start"], ge["end"]), (best["start"], best["end"])))
                type_total += 1
                if best["type"] == ge["type"]:
                    type_correct += 1
    per_type: dict[str, dict] = {}
    for t in set(tp_by_type) | set(fp_by_type) | set(fn_by_type):
        tp, fp, fn = tp_by_type[t], fp_by_type[t], fn_by_type[t]
        prec = tp / (tp + fp) if tp + fp else 0.0
        rec = tp / (tp + fn) if tp + fn else 0.0
        f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0.0
        per_type[t] = {"precision": prec, "recall": rec, "f1": f1, "support": tp + fn}
    tp = sum(tp_by_type.values())
    fp = sum(fp_by_type.values())
    fn = sum(fn_by_type.values())
    prec = tp / (tp + fp) if tp + fp else 0.0
    rec = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0.0
    return {
        "micro_precision": prec,
        "micro_recall": rec,
        "micro_f1": f1,
```

```python
# fine-tuning/scripts/eval_relation.py
"""Relation eval harness. Reports per-type P/R/F1, direction rate, exact-match."""
from __future__ import annotations

import argparse
import json
import sys
from collections import defaultdict
from pathlib import Path


def score(gold_rows: list[dict], pred_rows: dict) -> dict:
    tp_by_type: dict[str, int] = defaultdict(int)
    fp_by_type: dict[str, int] = defaultdict(int)
    fn_by_type: dict[str, int] = defaultdict(int)
    direction_correct = 0
    direction_total = 0
    exact_match = 0
    for g in gold_rows:
        p = pred_rows.get(g["id"], {}).get("relationships", [])
        gold_set = {(r["source"], r["target"], r["type"]) for r in g["relationships"]}
        pred_set = {(r["source"], r["target"], r["type"]) for r in p}
        for r in gold_set:
            if r in pred_set:
                tp_by_type[r[2]] += 1
            else:
                fn_by_type[r[2]] += 1
        for r in pred_set:
            if r not in gold_set:
                fp_by_type[r[2]] += 1
        # direction: for each gold (s,t,T), is predicted (s,t,T) or (t,s,T)?
        for (s, t, T) in gold_set:
            direction_total += 1
            if (s, t, T) in pred_set:
                direction_correct += 1
        if gold_set == pred_set:
            exact_match += 1
    per_type: dict[str, dict] = {}
    for t in set(tp_by_type) | set(fp_by_type) | set(fn_by_type):
        tp, fp, fn = tp_by_type[t], fp_by_type[t], fn_by_type[t]
        prec = tp / (tp + fp) if tp + fp else 0.0
        rec = tp / (tp + fn) if tp + fn else 0.0
        f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0.0
        per_type[t] = {"precision": prec, "recall": rec, "f1": f1, "support": tp + fn}
    tp, fp, fn = sum(tp_by_type.values()), sum(fp_by_type.values()), sum(fn_by_type.values())
    prec = tp / (tp + fp) if tp + fp else 0.0
    rec = tp / (tp + fn) if tp + fn else 0.0
    f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0.0
    return {
        "micro_precision": prec,
        "micro_recall": rec,
        "micro_f1": f1,
        "direction_rate": direction_correct / direction_total if direction_total else 0.0,
        "exact_match_rate": exact_match / len(gold_rows) if gold_rows else 0.0,
        "per_type": per_type,
    }


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--gold", type=Path, required=True)
    p.add_argument("--pred", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    gold = [json.loads(l) for l in a.gold.read_text(encoding="utf-8").splitlines() if l.strip()]
    pred = {json.loads(l)["id"]: json.loads(l) for l in a.pred.read_text(encoding="utf-8").splitlines() if l.strip()}
    metrics = score(gold, pred)
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(metrics, indent=2), encoding="utf-8")
    print(json.dumps({k: v for k, v in metrics.items() if k != "per_type"}, indent=2))
```

```python
# fine-tuning/scripts/eval_classifier.py
"""Memory classifier eval. Reports per-type accuracy, macro-F1, confusion matrix, reference recall."""
from __future__ import annotations

import argparse
import json
import sys
from collections import defaultdict
from pathlib import Path


def score(gold_rows: list[dict], pred_rows: dict) -> dict:
    classes = sorted({g["memory_type"] for g in gold_rows})
    conf: dict[str, dict[str, int]] = {c: {c2: 0 for c2 in classes} for c in classes}
    correct = 0
    total = 0
    tp_by_type: dict[str, int] = defaultdict(int)
    fp_by_type: dict[str, int] = defaultdict(int)
    fn_by_type: dict[str, int] = defaultdict(int)
    for g in gold_rows:
        gt = g["memory_type"]
        pt = pred_rows.get(g["id"], {}).get("memory_type") or pred_rows.get(g["id"], {}).get("label", "")
        conf.setdefault(gt, {}).setdefault(pt, 0)
        conf[gt][pt] = conf[gt].get(pt, 0) + 1
        total += 1
        if pt == gt:
            correct += 1
            tp_by_type[gt] += 1
        else:
            fn_by_type[gt] += 1
            fp_by_type[pt] += 1
    per_type: dict[str, dict] = {}
    for t in classes:
        tp, fp, fn = tp_by_type[t], fp_by_type[t], fn_by_type[t]
        prec = tp / (tp + fp) if tp + fp else 0.0
        rec = tp / (tp + fn) if tp + fn else 0.0
        f1 = 2 * prec * rec / (prec + rec) if prec + rec else 0.0
        per_type[t] = {"precision": prec, "recall": rec, "f1": f1, "support": tp + fn}
    macro_f1 = sum(v["f1"] for v in per_type.values()) / len(per_type) if per_type else 0.0
    return {
        "accuracy": correct / total if total else 0.0,
        "macro_f1": macro_f1,
        "reference_recall": per_type.get("reference", {}).get("recall", 0.0),
        "confusion_matrix": conf,
        "per_type": per_type,
    }


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--gold", type=Path, required=True)
    p.add_argument("--pred", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    gold = [json.loads(l) for l in a.gold.read_text(encoding="utf-8").splitlines() if l.strip()]
    pred = {json.loads(l)["id"]: json.loads(l) for l in a.pred.read_text(encoding="utf-8").splitlines() if l.strip()}
    metrics = score(gold, pred)
    a.out.parent.mkdir(parents=True, exist_ok=True)
    a.out.write_text(json.dumps(metrics, indent=2), encoding="utf-8")
    print(json.dumps({k: v for k, v in metrics.items() if k not in {"confusion_matrix", "per_type"}}, indent=2))
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
# fine-tuning/scripts/run_inference.py
"""Run inference from a trained checkpoint, producing a predictions JSONL with `id` preserved."""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path


def _load_gold(path: Path) -> list[dict]:
    return [json.loads(l) for l in path.read_text(encoding="utf-8").splitlines() if l.strip()]


def run_gliner(ckpt: str, gold: list[dict], entity_types: list[str], threshold: float = 0.5) -> list[dict]:
    from gliner import GLiNER
    model = GLiNER.from_pretrained(ckpt)
    preds: list[dict] = []
    for row in gold:
        out = model.predict_entities(row["text"], entity_types, threshold=threshold)
        entities = [{"name": e["text"], "type": e["label"], "start": e["start"], "end": e["end"]} for e in out]
        preds.append({"id": row["id"], "entities": entities})
    return preds


def run_rebel(ckpt: str, gold: list[dict]) -> list[dict]:
    from transformers import AutoTokenizer, AutoModelForSeq2SeqLM
    import torch
    tok = AutoTokenizer.from_pretrained(ckpt)
    model = AutoModelForSeq2SeqLM.from_pretrained(ckpt).to("cuda" if torch.cuda.is_available() else "cpu")
    preds: list[dict] = []
    for row in gold:
        ent_str = ", ".join(f"{e['name']} ({e['type']})" for e in row.get("entities", []))
        inp = f"{row['text']} | Entities: {ent_str}"
        enc = tok(inp, return_tensors="pt", max_length=512, truncation=True).to(model.device)
        out = model.generate(**enc, max_length=128, num_beams=5)
        decoded = tok.decode(out[0], skip_special_tokens=False)
        rels: list[dict] = []
        for tr in decoded.split("<triplet>"):
            tr = tr.strip()
            if not tr or tr == "NONE":
                continue
            parts = tr.split("<subj>")
            if len(parts) != 2:
                continue
            src = parts[0].strip()
            rest = parts[1].split("<obj>")
            if len(rest) != 2:
                continue
            tgt, typ = rest[0].strip(), rest[1].strip()
            if src and tgt and typ:
                rels.append({"source": src, "target": tgt, "type": typ})
        preds.append({"id": row["id"], "relationships": rels})
    return preds


def run_classifier(ckpt: str, gold: list[dict]) -> list[dict]:
    from transformers import AutoTokenizer, AutoModelForSequenceClassification
    import torch
    tok = AutoTokenizer.from_pretrained(ckpt)
    model = AutoModelForSequenceClassification.from_pretrained(ckpt).to("cuda" if torch.cuda.is_available() else "cpu")
    id2label = model.config.id2label
    preds: list[dict] = []
    for row in gold:
        enc = tok(row["text"], return_tensors="pt", truncation=True, max_length=256).to(model.device)
        with torch.no_grad():
            logits = model(**enc).logits
        label = id2label[int(logits.argmax(dim=-1).item())]
        preds.append({"id": row["id"], "memory_type": label})
    return preds


def main() -> int:
    p = argparse.ArgumentParser()
    p.add_argument("--task", required=True, choices=["entity", "relation", "classifier"])
    p.add_argument("--ckpt", required=True)
    p.add_argument("--gold", type=Path, required=True)
    p.add_argument("--out", type=Path, required=True)
    a = p.parse_args()
    gold = _load_gold(a.gold)
    if a.task == "entity":
        import yaml
        tax = yaml.safe_load(open(Path(__file__).resolve().parents[1] / "config" / "taxonomy_v2.yaml"))
        entity_types = [t for group in tax["entity_types"].values() for t in group]
        preds = run_gliner(a.ckpt, gold, entity_types)
    elif a.task == "relation":
        preds = run_rebel(a.ckpt, gold)
    else:
        preds = run_classifier(a.ckpt, gold)
    a.out.parent.mkdir(parents=True, exist_ok=True)
    with a.out.open("w", encoding="utf-8") as f:
        for r in preds:
            f.write(json.dumps(r) + "\n")
    print(f"wrote {len(preds)} predictions to {a.out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
