# darkmem Frontend Migration to darkraise-ui — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Rebuild darkmem's frontend to consume `darkraise-ui` for all UI components, theming, and layout, adopting TanStack Router and the template's dev tooling stack.

**Architecture:** Clean rebuild approach — scaffold a new `frontend/` from template conventions alongside the existing code (renamed to `frontend-legacy/`). Port pages incrementally in 4 phases, then swap directories and remove the legacy code.

**Tech Stack:** React 19, TanStack Router (file-based), TanStack Query, darkraise-ui (Radix UI + 6-axis theming + layout system), Zustand, Vite, Tailwind CSS 4

**Spec:** `docs/superpowers/specs/2026-04-07-darkraise-ui-migration-design.md`

```typescript
import { create } from "zustand"

interface AuthState {
  token: string | null
  isAuthenticated: boolean
  setToken: (token: string) => void
  logout: () => void
}

const STORAGE_KEY = "darkmem_token"

const getStoredToken = () => {
  try {
    return localStorage.getItem(STORAGE_KEY)
  } catch {
    return null
  }
}

export const useAuthStore = create<AuthState>((set) => ({
  token: getStoredToken(),
  isAuthenticated: !!getStoredToken(),

  setToken: (token: string) => {
    localStorage.setItem(STORAGE_KEY, token)
    set({ token, isAuthenticated: true })
  },

  logout: () => {
    localStorage.removeItem(STORAGE_KEY)
    set({ token: null, isAuthenticated: false })
  },
}))
```

```typescript
import { useAuthStore } from "@/features/auth"

const API_BASE = "/api/v1"

export async function apiFetch<T>(
  path: string,
  options: RequestInit = {},
): Promise<T> {
  const token = useAuthStore.getState().token
  const headers: Record<string, string> = {
    "Content-Type": "application/json",
    ...(options.headers as Record<string, string>),
  }
  if (token) {
    headers["Authorization"] = `Bearer ${token}`
  }

  const response = await fetch(`${API_BASE}${path}`, { ...options, headers })
  if (response.status === 401) {
    useAuthStore.getState().logout()
    window.location.href = "/login"
    return undefined as T
  }
  if (!response.ok) {
    const error = await response
      .json()
      .catch(() => ({ detail: response.statusText }))
    throw new Error(
      (error as { detail?: string }).detail || `API error: ${response.status}`,
    )
  }
  if (response.status === 204) return undefined as T
  return response.json() as Promise<T>
}

export async function checkSetupStatus(): Promise<{ initialized: boolean }> {
  const response = await fetch(`${API_BASE}/auth/status`)
  return response.json()
}

export async function login(password: string): Promise<string> {
  const response = await fetch(`${API_BASE}/auth/login`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ password }),
  })
  if (!response.ok) {
    const error = await response
      .json()
      .catch(() => ({ detail: response.statusText }))
    throw new Error((error as { detail?: string }).detail || "Login failed")
  }
  const data = await response.json()
  useAuthStore.getState().setToken(data.token)
  return data.token
}

export async function setup(password: string): Promise<string> {
  const response = await fetch(`${API_BASE}/auth/setup`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ password }),
  })
  if (!response.ok) {
    const error = await response
      .json()
      .catch(() => ({ detail: response.statusText }))
    throw new Error((error as { detail?: string }).detail || "Setup failed")
  }
  const data = await response.json()
  useAuthStore.getState().setToken(data.token)
  return data.token
}
```

- [ ] **Step 6: Create `frontend/src/lib/websocket.ts`**

Port from `frontend-legacy/src/hooks/useWebSocket.ts`. Extract as a standalone module that takes a `QueryClient`.

```typescript
import { useEffect, useRef, useCallback } from "react"
import { useQueryClient } from "@tanstack/react-query"
import { toast } from "darkraise-ui/components/sonner"

interface PipelineStatus {
  memory_id: string
  embed: string
  extract: string
  qdrant: string
  neo4j: string
  entities_count: number
  relations_count: number
  error: string | null
}

function showPipelineToast(event: string, data: PipelineStatus) {
  const id = data.memory_id.slice(0, 8)

  if (event === "pipeline.completed") {
    toast.success(`Memory ${id} processed`, {
      description: `${data.entities_count} entities, ${data.relations_count} relations extracted`,
    })
  } else if (event === "pipeline.partial") {
    const failed = [
      data.embed === "error" && "embedding",
      data.extract === "error" && "extraction",
      data.qdrant === "error" && "vector store",
      data.neo4j === "error" && "graph store",
    ]
      .filter(Boolean)
      .join(", ")
    toast.warning(`Memory ${id} partially processed`, {
      description: `Failed: ${failed}`,
    })
  } else if (event === "pipeline.failed") {
    toast.error(`Memory ${id} processing failed`, {
      description: data.error || "All pipeline steps failed",
    })
  }
}

export function useWebSocket() {
  const ws = useRef<WebSocket | null>(null)
  const retryCount = useRef<number>(0)
  const qc = useQueryClient()

  const connect = useCallback(() => {
    const protocol = window.location.protocol === "https:" ? "wss:" : "ws:"
    const url = `${protocol}//${window.location.host}/ws`
    ws.current = new WebSocket(url)

    ws.current.onmessage = (event) => {
      const msg = JSON.parse(event.data as string) as {
        event?: string
        data?: Record<string, unknown>
      }

      if (msg.event?.startsWith("pipeline.") && msg.data) {
        showPipelineToast(msg.event, msg.data as PipelineStatus)
        void qc.invalidateQueries({ queryKey: ["memories"] })
        void qc.invalidateQueries({ queryKey: ["stats"] })
        void qc.invalidateQueries({ queryKey: ["tags"] })
        void qc.invalidateQueries({ queryKey: ["pipelineLogs"] })
      } else if (msg.event?.startsWith("memory.")) {
        void qc.invalidateQueries({ queryKey: ["memories"] })
        void qc.invalidateQueries({ queryKey: ["stats"] })
        void qc.invalidateQueries({ queryKey: ["tags"] })
      }

      if (msg.event?.startsWith("repo.")) {
        void qc.invalidateQueries({ queryKey: ["repos"] })
        void qc.invalidateQueries({ queryKey: ["repoTasks"] })
        void qc.invalidateQueries({ queryKey: ["repoStats"] })

        if (msg.event === "repo.clone_completed") {
          toast.success(`Repository ${msg.data?.name} cloned`)
        } else if (msg.event === "repo.clone_failed") {
          toast.error(`Clone failed: ${msg.data?.error}`)
        } else if (msg.event === "repo.indexing_completed") {
          toast.success(
            `Indexed ${msg.data?.files_changed} files, ${msg.data?.symbols_indexed} symbols`,
          )
        } else if (msg.event === "repo.indexing_failed") {
          toast.error(`Indexing failed: ${msg.data?.error}`)
        }
      }

      if (msg.event?.startsWith("enrichment.")) {
        void qc.invalidateQueries({ queryKey: ["repoStats"] })
      }
    }

    ws.current.onopen = () => {
      retryCount.current = 0
    }

    ws.current.onclose = () => {
      const delay = Math.min(1000 * Math.pow(2, retryCount.current), 30000)
      retryCount.current++
      setTimeout(connect, delay)
    }
  }, [qc])

  useEffect(() => {
    connect()
    return () => ws.current?.close()
  }, [connect])
}
```
